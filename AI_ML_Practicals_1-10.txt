6 A: Missionaries and Cannibals
# State = (missionaries_left, cannibals_left, boat_side)
# 1 = boat on left, 0 = boat on right

from collections import deque

def safe(m, c):
    return (m == 0 or m >= c) and ((3-m) == 0 or (3-m) >= (3-c))

def missionaries_cannibals():
    start = (3, 3, 1)
    goal = (0, 0, 0)
    moves = [(1,0),(2,0),(0,1),(0,2),(1,1)]
    q = deque([(start, [start])])
    visited = {start}

    while q:
        state, path = q.popleft()
        m, c, boat = state

        if state == goal:
            return path

        for dm, dc in moves:
            if boat == 1:
                ns = (m-dm, c-dc, 0)
            else:
                ns = (m+dm, c+dc, 1)

            nm, nc, nb = ns
            if 0 <= nm <= 3 and 0 <= nc <= 3 and safe(nm,nc) and ns not in visited:
                visited.add(ns)
                q.append((ns, path+[ns]))

path = missionaries_cannibals()
print("Solution:")
for s in path:
    print(s)
